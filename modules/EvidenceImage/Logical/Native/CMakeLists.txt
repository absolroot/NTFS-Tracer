cmake_minimum_required(VERSION 3.21)
project(JournalDeskEwf C)
if(NOT MSVC OR NOT IS_DIRECTORY "${EWF_SOURCE}/libewf")
  message(FATAL_ERROR "Select MSVC and set EWF_SOURCE to the extracted libewf-python 20260924 source directory")
endif()
if(NOT IS_DIRECTORY "${ZLIB_SOURCE}")
  message(FATAL_ERROR "Set ZLIB_SOURCE to the extracted zlib 1.4 source directory")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_subdirectory("${ZLIB_SOURCE}" zlib EXCLUDE_FROM_ALL)
add_library(libewf SHARED)
set_property(TARGET libewf PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_include_directories(libewf PRIVATE "${EWF_SOURCE}/common" "${EWF_SOURCE}/include")
target_compile_definitions(libewf PRIVATE _CRT_SECURE_NO_WARNINGS UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 LIBEWF_DLL_EXPORT
  HAVE_ZLIB HAVE_ZLIB_COMPRESS2 HAVE_ZLIB_UNCOMPRESS HAVE_COMPRESS_BOUND HAVE_ZLIB_ADLER32)
target_compile_options(libewf PRIVATE /MP4)
foreach(lib libcerror libcthreads libcdata libcdatetime libclocale libcnotify libcsplit libuna libcfile libcpath libbfio libfcache libfdata libfdatetime libfguid libfvalue libhmac libcaes libewf)
  target_include_directories(libewf PRIVATE "${EWF_SOURCE}/${lib}")
  file(GLOB units "${EWF_SOURCE}/${lib}/*.c")
  target_sources(libewf PRIVATE ${units})
  if(NOT lib STREQUAL "libewf")
    string(TOUPPER "${lib}" flag)
    target_compile_definitions(libewf PRIVATE HAVE_LOCAL_${flag})
  endif()
endforeach()
target_link_libraries(libewf PRIVATE advapi32 ZLIB::ZLIBSTATIC)
